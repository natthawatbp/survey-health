  let draggedIssueId = null;

  window.dragStart = (e, id) => { 

    draggedIssueId = id; 

    e.dataTransfer.setData('text/plain', id); 

    setTimeout(() => e.target.classList.add('dragging'), 0);

  };

  window.allowDrop = (e) => { 

    e.preventDefault(); 

    e.currentTarget.classList.add('drag-over'); 

  };

  window.dragLeave = (e) => { 

    e.currentTarget.classList.remove('drag-over'); 

  };

  window.dropIssue = (e) => {

    e.preventDefault();

    e.currentTarget.classList.remove('drag-over');

    const catId = e.currentTarget.dataset.cat;

    if (!draggedIssueId || !catId) return;

    

    const item = document.getElementById('item-' + draggedIssueId);

    if(item) {

      item.classList.remove('dragging');

      e.currentTarget.querySelector('.kanban-items').appendChild(item);

    }

    

    task(async () => {

      await rpc('updateIssueCategory', token, draggedIssueId, catId);

      const rec = records.find(r => r.issue_id === draggedIssueId);

      if (rec) rec.main_category = catId;

      renderStaffBoardView(); 

    });

  };

  function getLocalVoterId() {

    let vid = localStorage.getItem('cv_voter_id');

    if (!vid) {

      vid = newId();

      localStorage.setItem('cv_voter_id', vid);

      localStorage.setItem('cv_votes_left', '3');

    }

    return vid;

  }

    votesLeft = parseInt(localStorage.getItem('cv_votes_left') || '3', 10);

    wallet();

    refreshBoard();

  }

  

  async function refreshBoard() {

    if (liveBusy || !config) return;

    liveBusy = true;

    try {

      liveBoard = await rpc('getLiveBoard');

      renderBoard();

      $('liveInfo').innerHTML = `อัปเดต ${new Date(liveBoard.updatedAt).toLocaleTimeString('th-TH')} • ผู้เข้าร่วมโหวต ${liveBoard.totals.participants} คน<br><span style="color:#258578">คัดเรื่องจาก 3 หมวดหมู่ที่มีคนพูดถึงมากที่สุด ได้แก่: ${liveBoard.top3Categories.map(catLabel).join(', ')}</span>`;

      $('liveInfo').textContent = 'อัปเดตไม่ได้ กดอัปเดตผลเพื่อลองอีกครั้ง';

    } finally {

      liveBusy = false;

    }

          <span class="tag auto" style="background:#e5f5eb; color:#23766d;">${esc(catText)}</span>

          <span class="voteCount">${n} ดาว · ${Math.round(n / total * 100)}%</span>

          <div class="bar"><i style="width:${n / total * 100}%"></i></div>

    if (!selectedVote) throw new Error('แตะเลือกประเด็นที่อยากโหวตก่อน');

    if (votesLeft < 1) throw new Error('คุณใช้โควตาครบ 3 ดาวแล้ว ขอบคุณที่ร่วมกิจกรรม');

    votesLeft = r.remaining;

    localStorage.setItem('cv_votes_left', votesLeft.toString());

    voteRequest = '';

    $('sendVote').disabled = true;

    selectedVote = '';

    wallet();

    await refreshBoard();
